cmake_minimum_required(VERSION 3.18)
project(nosmai_zoom_frames LANGUAGES CXX)
set(CLASSES "${CMAKE_CURRENT_SOURCE_DIR}/../../../../ios/Classes")
file(GLOB YUV_SOURCES CONFIGURE_DEPENDS "${CLASSES}/Yuv/source/*.cc")
if(ANDROID_ABI STREQUAL "arm64-v8a")
    file(GLOB NEON64 "${CLASSES}/Yuv/source/*_neon64.cc")
    set_source_files_properties(${NEON64} PROPERTIES COMPILE_OPTIONS "-march=armv8.2-a+dotprod+i8mm")
endif()
add_library(nosmai_zoom_frames SHARED ZoomFrames.cpp "${CLASSES}/Native/FrameFit.cc" ${YUV_SOURCES})
target_include_directories(nosmai_zoom_frames PRIVATE "${CLASSES}/Native" "${CLASSES}/Yuv/include")
target_compile_definitions(nosmai_zoom_frames PRIVATE LIBYUV_DISABLE_SVE LIBYUV_DISABLE_SME)
target_compile_options(nosmai_zoom_frames PRIVATE -O2 -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(nosmai_zoom_frames PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
