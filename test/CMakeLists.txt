cmake_minimum_required(VERSION 3.18)
project(zoom_frame_tests LANGUAGES CXX)
enable_testing()
set(CLASSES "${CMAKE_CURRENT_SOURCE_DIR}/../ios/Classes")
file(GLOB YUV_SOURCES "${CLASSES}/Yuv/source/*.cc")
add_executable(frame_fit_test frame_fit_test.cc "${CLASSES}/Native/FrameFit.cc" ${YUV_SOURCES})
target_include_directories(frame_fit_test PRIVATE "${CLASSES}/Native" "${CLASSES}/Yuv/include")
target_compile_definitions(frame_fit_test PRIVATE LIBYUV_DISABLE_SME LIBYUV_DISABLE_SVE)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  file(GLOB NEON64 "${CLASSES}/Yuv/source/*_neon64.cc")
  set_source_files_properties(${NEON64} PROPERTIES COMPILE_OPTIONS "-march=armv8.2-a+dotprod+i8mm")
endif()
target_compile_options(frame_fit_test PRIVATE -UNDEBUG)
add_test(NAME frame_fit COMMAND frame_fit_test)
